class Solution {
public:
    int myAtoi(string s) {
        int i = 0;
        int n = s.size();

        while (i < n && s[i] == ' ') {
            ++i;
        }

        int sign = 1;
        if (i < n && (s[i] == '+' || s[i] == '-')) {
            if (s[i] == '-') sign = -1;
            ++i;
        }

        int result = 0;
        int limit = (sign == 1) ? INT_MAX : INT_MIN;

        while (i < n && s[i] >= '0' && s[i] <= '9') {
            int digit = s[i] - '0';

            if (sign == 1) {
                if (result > INT_MAX / 10 ||
                    (result == INT_MAX / 10 && digit > 7)) {
                    return INT_MAX;
                }
                result = result * 10 + digit;
            } else {
                if (result < INT_MIN / 10 ||
                    (result == INT_MIN / 10 && digit > 8)) {
                    return INT_MIN;
                }
                result = result * 10 - digit;
            }
            ++i;
        }

        return result;
    }
};
